% =====================================================================
%  templates/tailored-cv.tex : designed tailored-CV skeleton
%
%  Usage (tailor skill):
%    1. Copy to applications/<company>-<role>/<name>-cv-<company>.tex
%       (<name> is the file-name slug in profile.md: every file an
%       employer receives leads with the user's name, because a file
%       called cv-acme.pdf says nothing about whose CV it is once it is
%       in a recruiter's downloads folder).
%    2. Replace every {{...}} placeholder. Content rules (CLAUDE.md):
%       - every claim comes from MASTER_CV.md: reorder, select, rephrase,
%         never invent; every number verbatim, durations included
%       - no em dashes anywhere; date ranges use "--" (en dash)
%       - escape LaTeX specials: \& \% \#
%    3. Build with XeLaTeX (fontspec, NOT pdflatex), two passes:
%         xelatex <name>-cv-<company>.tex
%    4. Read the built PDF once as a page. A title colliding with its
%       date, or a heading stranded at the foot of a page, never shows
%       up in the log.
%
%  Target length: 1-2 pages, and two is the target for most people: length
%  should track output. Cut projects before cutting experience.
%  This file is a skeleton; it is not meant to compile until the
%  placeholders are replaced.
%
%  FONTS: EB Garamond and Playfair Display ship with both TeX Live (the
%  ebgaramond and playfair packages) and MiKTeX, so they resolve from the
%  distribution's own font tree with no system font installation. If your
%  distribution is missing them, comment out the fontspec block below and
%  uncomment the fallback: the layout survives, it just looks plainer.
% =====================================================================
\documentclass[11pt]{article}

\usepackage[a4paper,top=1.4cm,bottom=1.3cm,left=1.7cm,right=1.7cm,footskip=0.5cm]{geometry}

% ---- Fonts -----------------------------------------------------------
\usepackage{fontspec}
\setmainfont{EBGaramond}[
  Extension=.otf, UprightFont=*-Regular, BoldFont=*-Bold,
  ItalicFont=*-Italic, BoldItalicFont=*-BoldItalic ]
\newfontfamily\displayfont{PlayfairDisplay}[
  Extension=.otf, UprightFont=*-Regular, BoldFont=*-Bold,
  ItalicFont=*-Italic, BoldItalicFont=*-BoldItalic ]

% ---- Fallback if those fonts are not installed -----------------------
% \usepackage{fontspec}

% ---- Colours ---------------------------------------------------------
\usepackage{xcolor}
\definecolor{accent}{HTML}{2A5C8A}     % change this one line to restyle
\definecolor{datecolor}{HTML}{4A6E8A}
\definecolor{bodytext}{HTML}{2B2B2B}

% ---- Packages --------------------------------------------------------
\usepackage{fontawesome5}
\usepackage{enumitem}
\usepackage[hidelinks]{hyperref}
\usepackage{microtype}

% ---- Global layout ---------------------------------------------------
\color{bodytext}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}
\pagestyle{empty}
\linespread{1.05}
\setlength{\emergencystretch}{2em}

% ---- Keep a heading with what follows it -----------------------------
% Break the page early if fewer than five lines remain, so a section
% heading is never stranded at the foot of a page. Plain TeX on purpose:
% no extra package to install.
\newcommand{\keepwithnext}{\par\ifdim\dimexpr\pagegoal-\pagetotal\relax<5\baselineskip\newpage\fi}

% ---- Section heading -------------------------------------------------
% The rule is drawn with \hrule in vertical mode, not an inline \rule: an
% inline rule sits on the baseline of a fresh text line, which puts it a
% full line below the heading and crowds the body text.
\newcommand{\cvsection}[1]{%
  \keepwithnext\vspace{8pt}%
  {\displayfont\large\bfseries\color{accent}\MakeUppercase{#1}}\par
  \nointerlineskip\vspace{2.5pt}%
  {\color{accent}\hrule height 1pt}%
  \nointerlineskip\vspace{5pt}}

% ---- Entry: bold title + right-aligned date, italic org line ---------
% \cventry{Title}{Date}{Organisation}
% The leading \strut in each box, and \textcolor rather than a bare \color
% switch, keep the two first baselines level: a colour whatsit at the top
% of a [t] minipage lets the date drift below the title. The trailing
% \strut gives a wrapped title's last line its full depth, so the
% organisation line below it keeps a normal line distance.
\newcommand{\cventry}[3]{%
  \noindent
  \begin{minipage}[t]{0.72\linewidth}\raggedright\strut\textbf{#1}\strut\end{minipage}%
  \hfill
  \begin{minipage}[t]{0.26\linewidth}\raggedleft\small\strut\textcolor{datecolor}{#2}\end{minipage}\par
  {\itshape #3}\par\vspace{1pt}}

% ---- Tight bullet list ----------------------------------------------
\newlist{cvitems}{itemize}{1}
\setlist[cvitems]{leftmargin=1.35em,labelsep=0.5em,itemsep=1.5pt,topsep=2pt,parsep=0pt,
  label={\small\color{accent}\textbullet}}

% ---- One contact item: \contact{\faIcon}{text} ----------------------
\newcommand{\contact}[2]{{\color{accent}#1}~#2}
